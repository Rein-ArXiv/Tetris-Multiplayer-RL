cmake_minimum_required(VERSION 3.18)
project(study_inference LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
set(ORT_ROOT "" CACHE PATH "Extracted ONNX Runtime CPU SDK with include and libraries")
set(ORT_LIBRARY "" CACHE FILEPATH "ONNX Runtime library for the target OS and architecture")
find_path(ORT_INCLUDE_DIR onnxruntime_cxx_api.h HINTS "${ORT_ROOT}/include" REQUIRED)
if(NOT ORT_LIBRARY)
    find_library(STUDY_ORT_LIBRARY NAMES onnxruntime HINTS "${ORT_ROOT}/lib"
        "${ORT_ROOT}/lib/linux-x64" "${ORT_ROOT}/lib/linux-aarch64"
        "${ORT_ROOT}/lib/osx-universal2" "${ORT_ROOT}/lib/win-x64" REQUIRED)
    set(ORT_LIBRARY "${STUDY_ORT_LIBRARY}")
endif()
add_library(study_policy ../bot/onnx_policy.cpp)
target_compile_features(study_policy PUBLIC cxx_std_17)
target_include_directories(study_policy PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/.." PRIVATE "${ORT_INCLUDE_DIR}")
target_link_libraries(study_policy PUBLIC "${ORT_LIBRARY}")
add_executable(onnx_probe ../tools/onnx_probe.cpp)
target_include_directories(onnx_probe PRIVATE "${ORT_INCLUDE_DIR}")
target_link_libraries(onnx_probe PRIVATE study_policy)
add_executable(inference_contract ../tests/inference_contract.cpp)
target_link_libraries(inference_contract PRIVATE study_policy)
add_executable(policy_choice ../tests/policy_choice.cpp)
target_include_directories(policy_choice PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_features(policy_choice PRIVATE cxx_std_17)
enable_testing()
add_test(NAME policy_choice COMMAND policy_choice)

add_executable(onnx_paced_probe ../tools/onnx_paced_probe.cpp)
target_link_libraries(onnx_paced_probe PRIVATE study_policy)

add_executable(character_probe ../tools/character_probe.cpp)
target_link_libraries(character_probe PRIVATE study_policy)
