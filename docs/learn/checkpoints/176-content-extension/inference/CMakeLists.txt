cmake_minimum_required(VERSION 3.18)
project(study_inference LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
set(ORT_ROOT "" CACHE PATH "ONNX Runtime SDK in include/ and lib/<platform>/ layout")
include("${CMAKE_CURRENT_SOURCE_DIR}/../dependencies/OnnxRuntime.cmake")
tetris_import_onnxruntime("${ORT_ROOT}")
add_library(study_policy STATIC ../bot/onnx_policy.cpp)
target_compile_features(study_policy PUBLIC cxx_std_17)
target_include_directories(study_policy PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(study_policy PRIVATE Tetris::OnnxRuntime)
add_executable(onnx_probe ../tools/onnx_probe.cpp)
target_link_libraries(onnx_probe PRIVATE study_policy Tetris::OnnxRuntime)
add_executable(inference_contract ../tests/inference_contract.cpp)
target_link_libraries(inference_contract PRIVATE study_policy)
add_executable(policy_choice ../tests/policy_choice.cpp)
target_include_directories(policy_choice PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_features(policy_choice PRIVATE cxx_std_17)
enable_testing()
add_test(NAME policy_choice COMMAND policy_choice)

add_executable(onnx_paced_probe ../tools/onnx_paced_probe.cpp)
target_link_libraries(onnx_paced_probe PRIVATE study_policy)

add_executable(character_probe ../tools/character_probe.cpp)
target_link_libraries(character_probe PRIVATE study_policy)

add_executable(policy_match_probe ../tools/policy_match_probe.cpp)
target_link_libraries(policy_match_probe PRIVATE study_policy)

# A small diagnostic links and starts the real SDK without a model file.
add_executable(dependency_probe ../tools/dependency_probe.cpp)
target_link_libraries(dependency_probe PRIVATE Tetris::OnnxRuntime)
target_compile_features(dependency_probe PRIVATE cxx_std_17)
add_test(NAME dependency_probe COMMAND dependency_probe)

# Local relocation exercise: the installed tree owns its Runtime files.
set(_study_packaged_targets dependency_probe onnx_probe policy_match_probe)
foreach(_study_target IN LISTS _study_packaged_targets)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        set_target_properties(${_study_target} PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
    elseif(APPLE)
        set_target_properties(${_study_target} PROPERTIES INSTALL_RPATH "@loader_path/../lib")
    endif()
    set_target_properties(${_study_target} PROPERTIES INSTALL_RPATH_USE_LINK_PATH FALSE)
endforeach()
install(TARGETS ${_study_packaged_targets} RUNTIME DESTINATION bin)
get_target_property(_study_ort_runtime Tetris::OnnxRuntime IMPORTED_LOCATION)
get_filename_component(_study_ort_dir "${_study_ort_runtime}" DIRECTORY)
if(WIN32)
    install(DIRECTORY "${_study_ort_dir}/" DESTINATION bin FILES_MATCHING PATTERN "*.dll")
else()
    install(DIRECTORY "${_study_ort_dir}/" DESTINATION lib
        FILES_MATCHING PATTERN "*.so*" PATTERN "*.dylib*")
endif()
